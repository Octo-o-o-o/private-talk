### 1.1 新建 `src-tauri/src/image_gen.rs`

内容：

**类型定义：**

```rust
use serde::{Deserialize, Serialize};

/// 统一的图片生成请求参数
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ImageGenerationRequest {
    pub prompt: String,
    pub aspect_ratio: Option<String>,   // "1:1", "16:9", "9:16", "4:3", "3:4"
    pub quality: Option<String>,        // "standard" | "hd"
    pub count: Option<u8>,              // 1..=4
    pub background: Option<String>,     // "auto" | "transparent" | "opaque"
    pub reference_images: Vec<ReferenceImage>,
}

#[derive(Debug, Clone)]
pub struct ReferenceImage {
    pub data: Vec<u8>,
    pub mime_type: String,
}

/// 单张生成结果
pub struct GeneratedImage {
    pub data: Vec<u8>,
    pub mime_type: String,  // "image/png"
}

/// 生成结果
pub struct ImageGenerationResult {
    pub images: Vec<GeneratedImage>,
    pub revised_prompt: Option<String>,
}
```

**参数映射函数：**

```rust
/// aspect_ratio → OpenAI size string
fn map_size(aspect_ratio: &str, model: &str) -> String {
    let is_dalle3 = model.contains("dall-e-3");
    match aspect_ratio {
        "16:9" => if is_dalle3 { "1792x1024" } else { "1536x1024" }.to_string(),
        "9:16" => if is_dalle3 { "1024x1792" } else { "1024x1536" }.to_string(),
        _ => "1024x1024".to_string(), // 1:1, 4:3, 3:4 等 fallback
    }
}

/// quality → OpenAI quality string
fn map_quality(quality: &str, model: &str) -> String {
    let is_dalle3 = model.contains("dall-e-3");
    match quality {
        "hd" => if is_dalle3 { "hd" } else { "high" }.to_string(),
        _ => if is_dalle3 { "standard" } else { "medium" }.to_string(),
    }
}
```

**文生图（无参考图）：**

```rust
/// POST {base_url}/images/generations
pub async fn generate_images(
    base_url: &str,
    api_key: &str,
    model: &str,
    request: &ImageGenerationRequest,
) -> Result<ImageGenerationResult, String>
```

实现要点：
- URL: `format!("{}/images/generations", base_url.trim_end_matches('/'))`
- 请求 body 为 JSON：`{ model, prompt, n, size, quality, background, output_format: "png" }`
  - `output_format` 为 `"png"` 时返回 `b64_json`
  - DALL-E 3 用 `response_format: "b64_json"` 代替 `output_format`
- 响应解析：`data[].b64_json` 或 `data[].url`
  - 如果是 URL，用 reqwest 下载为 bytes
- 超时：120 秒
- Bearer auth（复用现有 `with_optional_bearer` 模式）

**图编辑（有参考图）：**

```rust
/// POST {base_url}/images/edits (multipart/form-data)
pub async fn edit_images(
    base_url: &str,
    api_key: &str,
    model: &str,
    request: &ImageGenerationRequest,
) -> Result<ImageGenerationResult, String>
```

实现要点：
- URL: `format!("{}/images/edits", base_url.trim_end_matches('/'))`
- 请求格式：`multipart/form-data`
  - `model`: text field
  - `prompt`: text field
  - `image`: file part（第一张参考图）
  - `n`: text field
  - `size`: text field
- 响应格式与 generations 相同

**`/img` 参数解析：**

```rust
/// 解析 "/img prompt --ratio 16:9 --quality hd" 格式
pub fn parse_img_command(
    content: &str,
    default_ratio: &str,
    default_quality: &str,
) -> Result<ImageGenerationRequest, String>
```

解析规则：
1. 去掉 `/img ` 前缀
2. 从末尾向前扫描 `--key value` 和 `-k value` 参数
3. 支持的 key: `ratio`/`r`, `quality`/`q`, `count`/`n`, `bg`
4. 剩余部分为 prompt（trim）
5. prompt 不能为空

### 4.2 实现 `handle_image_generation()`

可以放在同一个文件中，或新建 `src-tauri/src/commands/image_gen.rs`。

推荐新建文件，保持 chat.rs 聚焦。

**新建 `src-tauri/src/commands/image_gen.rs`：**

```rust
use crate::attachments;
use crate::db::DbState;
use crate::image_gen::{self, ImageGenerationRequest, parse_img_command};
use serde::Serialize;
use tauri::{Emitter, State};

#[derive(Clone, Serialize)]
pub struct ImageGenStatusPayload {
    conversation_id: String,
    phase: String,       // "generating" | "saving" | "done" | "failed"
    message: Option<String>,
}

/// Handle /img command within send_message flow
pub async fn handle_image_generation(
    app: tauri::AppHandle,
    db: &State<'_, DbState>,
    conversation_id: &str,
    content: &str,
    user_msg_id: &str,
    prepared_attachments: &[crate::attachments::PreparedAttachment],
) -> Result<(), String> {
    // 1. 读取 image_gen_config
    let config = {
        let conn = db.0.lock().map_err(|e| e.to_string())?;
        let raw: Option<String> = conn.query_row(
            "SELECT value FROM settings WHERE key = 'image_gen_config'",
            [],
            |row| row.get(0),
        ).ok();
        raw.and_then(|s| serde_json::from_str::<serde_json::Value>(&s).ok())
    };

    let config = config.ok_or("Image generation is not configured. Please set it up in Settings.")?;
    let enabled = config.get("enabled").and_then(|v| v.as_bool()).unwrap_or(false);
    if !enabled {
        return Err("Image generation is disabled. Enable it in Settings.".to_string());
    }

    let provider_id = config.get("provider_id").and_then(|v| v.as_str())
        .ok_or("Image generation provider not configured")?;
    let model = config.get("model").and_then(|v| v.as_str())
        .ok_or("Image generation model not configured")?;
    let default_ratio = config.get("default_aspect_ratio")
        .and_then(|v| v.as_str()).unwrap_or("1:1");
    let default_quality = config.get("default_quality")
        .and_then(|v| v.as_str()).unwrap_or("standard");
    let max_n: u8 = config.get("max_images_per_request")
        .and_then(|v| v.as_u64()).unwrap_or(4) as u8;

    // 2. 解析 /img 参数
    let mut request = parse_img_command(content, default_ratio, default_quality)?;
    if let Some(n) = request.count {
        if n > max_n { request.count = Some(max_n); }
    }

    // 3. 加载 provider
    let (base_url, api_key) = {
        let conn = db.0.lock().map_err(|e| e.to_string())?;
        conn.query_row(
            "SELECT base_url, api_key FROM providers WHERE id = ?1",
            rusqlite::params![provider_id],
            |row| Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?)),
        ).map_err(|e| format!("Image provider not found: {}", e))?
    };

    // 4. 读取参考图（如有图片附件）
    for att in prepared_attachments {
        if att.file_type == "image" {
            let data = std::fs::read(&att.file_path)
                .map_err(|e| format!("Failed to read reference image: {}", e))?;
            request.reference_images.push(image_gen::ReferenceImage {
                data,
                mime_type: att.mime_type.clone(),
            });
        }
    }

    // 5. 发出进度事件
    let _ = app.emit("image-gen-status", ImageGenStatusPayload {
        conversation_id: conversation_id.to_string(),
        phase: "generating".to_string(),
        message: None,
    });

    // 6. 调用 API
    let result = if request.reference_images.is_empty() {
        image_gen::generate_images(&base_url, &api_key, model, &request).await
    } else {
        image_gen::edit_images(&base_url, &api_key, model, &request).await
    };

    let result = match result {
        Ok(r) => r,
        Err(e) => {
            let _ = app.emit("image-gen-status", ImageGenStatusPayload {
                conversation_id: conversation_id.to_string(),
                phase: "failed".to_string(),
                message: Some(e.clone()),
            });
            return Err(e);
        }
    };

    // 7. 保存结果
    let _ = app.emit("image-gen-status", ImageGenStatusPayload {
        conversation_id: conversation_id.to_string(),
        phase: "saving".to_string(),
        message: None,
    });

    let n = result.images.len();
    let has_ref = !request.reference_images.is_empty();

    // 创建 assistant 消息
    let assistant_msg_id = uuid::Uuid::new_v4().to_string();
    let now = chrono::Utc::now().format("%Y-%m-%d %H:%M:%S").to_string();
    let assistant_content = if has_ref {
        format!("已基于参考图生成 {} 张图片。\nGenerated {} image(s) from reference.", n, n)
    } else {
        format!("已生成 {} 张图片。\nGenerated {} image(s).", n, n)
    };

    let app_data_dir = app.path().app_data_dir()
        .map_err(|e| format!("Failed to get app data dir: {}", e))?;

    {
        let conn = db.0.lock().map_err(|e| e.to_string())?;

        // 保存 assistant 消息
        conn.execute(
            "INSERT INTO messages (id, conversation_id, role, content, created_at) VALUES (?1, ?2, 'assistant', ?3, ?4)",
            rusqlite::params![assistant_msg_id, conversation_id, assistant_content, now],
        ).map_err(|e| e.to_string())?;

        // 保存每张图片为附件
        let metadata = serde_json::json!({
            "source": "generated",
            "generation": {
                "provider_id": provider_id,
                "model": model,
                "prompt": request.prompt,
                "revised_prompt": result.revised_prompt,
                "aspect_ratio": request.aspect_ratio,
                "quality": request.quality,
                "count": n,
                "background": request.background,
            }
        });

        for gen_image in &result.images {
            attachments::save_generated_image(
                &app_data_dir,
                &gen_image.data,
                &gen_image.mime_type,
                &assistant_msg_id,
                metadata.clone(),
                &conn,
            )?;
        }
    }

    // 8. 发出完成事件
    let _ = app.emit("image-gen-status", ImageGenStatusPayload {
        conversation_id: conversation_id.to_string(),
        phase: "done".to_string(),
        message: None,
    });

    // 复用现有 chat-stream-done 事件，让前端刷新消息列表
    let _ = app.emit("chat-stream-done", serde_json::json!({
        "conversation_id": conversation_id,
        "message_id": assistant_msg_id,
        "full_content": assistant_content,
    }));

    Ok(())
}
```
